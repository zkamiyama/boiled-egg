cmake_minimum_required(VERSION 3.20)
project(relocated_sdk_consumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT EXPECT_LINKAGE MATCHES "^(shared|static)$" OR NOT EXPECT_SPECTRAL MATCHES "^[01]$")
  message(FATAL_ERROR "Explicit EXPECT_LINKAGE and EXPECT_SPECTRAL are required")
endif()
find_package(boiled_egg CONFIG REQUIRED)
get_target_property(kind boiled_egg::boiled_egg TYPE)
get_target_property(defs boiled_egg::boiled_egg INTERFACE_COMPILE_DEFINITIONS)
if(EXPECT_LINKAGE STREQUAL "static")
  if(NOT kind STREQUAL "STATIC_LIBRARY" OR NOT "BOILED_EGG_STATIC=1" IN_LIST defs)
    message(FATAL_ERROR "Static SDK must export its static-client usage requirement")
  endif()
else()
  if(NOT kind STREQUAL "SHARED_LIBRARY" OR "BOILED_EGG_STATIC=1" IN_LIST defs)
    message(FATAL_ERROR "Shared SDK must not export the static-client definition")
  endif()
endif()
if("BOILED_EGG_BUILDING_LIBRARY=1" IN_LIST defs)
  message(FATAL_ERROR "Private DLL-export definition leaked to clients")
endif()
enable_testing()
# Reuse existing installed ABI/state/ramp/RAII contracts; no internal headers.
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../backend_install" legacy)
add_executable(relocated_pcm_c stream.c)
set_target_properties(relocated_pcm_c PROPERTIES LINKER_LANGUAGE CXX)
add_executable(relocated_pcm_cpp wrapper.cpp)
foreach(t relocated_pcm_c relocated_pcm_cpp)
  target_link_libraries(${t} PRIVATE boiled_egg::boiled_egg)
  add_test(NAME ${t} COMMAND ${t})
  set_tests_properties(${t} PROPERTIES TIMEOUT 30)
endforeach()
