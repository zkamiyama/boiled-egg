cmake_minimum_required(VERSION 3.20)
project(installed_backend_contract LANGUAGES C CXX)
find_package(boiled_egg CONFIG REQUIRED)
set(EXPECT_SPECTRAL 0 CACHE STRING "Require the explicitly built spectral preview")
add_executable(backend_c consumer.c)
set_target_properties(backend_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES LINKER_LANGUAGE CXX)
add_executable(backend_cpp consumer.cpp)
foreach(target backend_c backend_cpp)
  target_link_libraries(${target} PRIVATE boiled_egg::boiled_egg)
  target_compile_definitions(${target} PRIVATE EXPECT_SPECTRAL=${EXPECT_SPECTRAL})
endforeach()
enable_testing()
add_test(NAME installed_backend_c COMMAND backend_c)
add_test(NAME installed_backend_cpp COMMAND backend_cpp)
