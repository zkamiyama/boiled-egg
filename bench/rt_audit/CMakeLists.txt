cmake_minimum_required(VERSION 3.20)
project(boiled_egg_rt_audit LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This diagnostic uses Linux clocks and absolute sleeps")
endif()
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(DEFAULT_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BOILED_EGG_SOURCE_ROOT "${DEFAULT_SOURCE_ROOT}" CACHE PATH "SDK source tree")
option(RT_AUDIT_RESEARCH "Use the separate research host API" OFF)
if(RT_AUDIT_RESEARCH)
  set(BOILED_EGG_PV_RT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BOILED_EGG_PV_RT_BUILD_CLI OFF CACHE BOOL "" FORCE)
  add_subdirectory("${BOILED_EGG_SOURCE_ROOT}/research/cpp_pv_rt" backend)
else()
  set(BOILED_EGG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BOILED_EGG_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(BOILED_EGG_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  add_subdirectory("${BOILED_EGG_SOURCE_ROOT}" backend)
endif()
add_executable(rt_audit_probe probe.cpp)
target_compile_options(rt_audit_probe PRIVATE -Wall -Wextra -Wpedantic)
if(RT_AUDIT_RESEARCH)
  target_compile_definitions(rt_audit_probe PRIVATE BOILED_EGG_AUDIT_RESEARCH=1)
  target_link_libraries(rt_audit_probe PRIVATE boiled_egg_research_host)
else()
  target_link_libraries(rt_audit_probe PRIVATE boiled_egg::boiled_egg)
endif()
add_executable(rt_audit_protocol_test test_protocol.cpp)
enable_testing()
add_test(NAME protocol_calibration COMMAND rt_audit_protocol_test)
