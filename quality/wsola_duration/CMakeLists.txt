cmake_minimum_required(VERSION 3.20)
project(wsola_duration_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(boiled_egg CONFIG REQUIRED)
add_executable(duration_check check.cpp)
target_link_libraries(duration_check PRIVATE boiled_egg::boiled_egg)
add_executable(duration_cost cost.cpp)
target_link_libraries(duration_cost PRIVATE boiled_egg::boiled_egg)
enable_testing()
foreach(mode repro edges partitions dynamic backpressure)
  add_test(NAME duration_${mode} COMMAND duration_check ${mode})
  set_tests_properties(duration_${mode} PROPERTIES TIMEOUT 90)
endforeach()
