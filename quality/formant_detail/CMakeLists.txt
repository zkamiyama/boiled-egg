cmake_minimum_required(VERSION 3.20)
project(formant_detail_controls LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "20 or 23")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(BOILED_EGG_BUILD_TESTS OFF CACHE BOOL "")
set(BOILED_EGG_BUILD_TOOLS OFF CACHE BOOL "")
set(BOILED_EGG_BUILD_BENCH OFF CACHE BOOL "")
set(BOILED_EGG_ENABLE_EXPERIMENTAL_SPECTRAL ON CACHE BOOL "")
if(BOILED_EGG_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(BOILED_EGG_THREAD_SANITIZE)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_subdirectory("${ROOT}" sdk EXCLUDE_FROM_ALL)
add_library(formant_direct STATIC "${ROOT}/src/experimental/pv/pv_rt.cpp" "${ROOT}/src/experimental/pv/fft.cpp")
target_include_directories(formant_direct PUBLIC "${ROOT}/src/experimental/pv" "${ROOT}/include")
find_package(Threads REQUIRED)
add_executable(formant_controls controls.cpp)
target_link_libraries(formant_controls PRIVATE boiled_egg::boiled_egg formant_direct Threads::Threads)
add_executable(formant_c_consumer consumer.c)
set_target_properties(formant_c_consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(formant_c_consumer PRIVATE boiled_egg::boiled_egg)
add_executable(formant_render render.cpp)
target_link_libraries(formant_render PRIVATE boiled_egg::boiled_egg)
enable_testing()
if(BOILED_EGG_THREAD_SANITIZE)
 add_test(NAME detail_parallel COMMAND formant_controls parallel)
else()
 foreach(name validation stream realtime events noalloc parallel nonfinite)
  add_test(NAME detail_${name} COMMAND formant_controls ${name})
 endforeach()
 add_test(NAME detail_c11 COMMAND formant_c_consumer)
endif()
