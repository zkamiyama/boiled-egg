cmake_minimum_required(VERSION 3.20)
project(wsola_cost_controls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(overlap_cost ../../tests/test_overlap_cost.cpp ../../src/engine.cpp)
target_include_directories(overlap_cost PRIVATE ../../src ../../include)
add_test(NAME overlap_cost_exact COMMAND overlap_cost)
set_tests_properties(overlap_cost_exact PROPERTIES TIMEOUT 60)
option(WSOLA_COST_BUILD_BENCH "Linux public-ABI timing client, not a CTest performance gate" OFF)
if(WSOLA_COST_BUILD_BENCH)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The thread-clock benchmark currently requires Linux")
  endif()
  find_package(boiled_egg CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(wsola_cost_bench bench.cpp)
  target_link_libraries(wsola_cost_bench PRIVATE boiled_egg::boiled_egg Threads::Threads)
endif()
