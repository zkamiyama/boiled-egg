cmake_minimum_required(VERSION 3.20)
project(streaming_budget_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(boiled_egg CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(streaming_budget test.cpp)
target_link_libraries(streaming_budget PRIVATE boiled_egg::boiled_egg Threads::Threads)
if(NOT MSVC)
  target_compile_options(streaming_budget PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
foreach(mode repro corners partition dynamic pressure invalid noalloc)
  add_test(NAME streaming_budget_${mode} COMMAND streaming_budget ${mode})
  set_tests_properties(streaming_budget_${mode} PROPERTIES TIMEOUT 120)
endforeach()
