cmake_minimum_required(VERSION 3.20)
project(pv_ring_controls LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "20 or 23")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
# Existing public/direct, reset, events, noalloc and C11 controls are not copied.
add_subdirectory(../formant_detail detail)
add_executable(ring_controls controls.cpp)
target_include_directories(ring_controls PRIVATE ../../src/experimental/pv)
if(BOILED_EGG_SANITIZE)
 target_compile_options(ring_controls PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(ring_controls PRIVATE -fsanitize=address,undefined)
endif()
foreach(name boundaries exhaustive random capacities wrapped_storage)
 add_test(NAME ring_${name} COMMAND ring_controls ${name})
endforeach()
