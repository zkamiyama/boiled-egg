cmake_minimum_required(VERSION 3.20)
project(pv_fft_stage_controls LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "20 or 23")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BE_FFT_REFERENCE_SOURCE "" CACHE PATH "Immutable d4093d87 source snapshot")
foreach(ext cpp hpp)
  if(NOT EXISTS "${BE_FFT_REFERENCE_SOURCE}/src/experimental/pv/fft.${ext}")
    message(FATAL_ERROR "Explicit immutable reference source required")
  endif()
  file(SHA256 "${BE_FFT_REFERENCE_SOURCE}/src/experimental/pv/fft.${ext}" digest)
  if(ext STREQUAL "cpp")
    set(expected e29498183d75e15856a5deedadbf152b6948b19119f9fc7186577839d1c1adaf)
  else()
    set(expected fd734ac760ced3bf87db759013aff6d7d4b9ec166a63dd763bff0446fc91a105)
  endif()
  if(NOT digest STREQUAL expected)
    message(FATAL_ERROR "Reference FFT does not match d4093d87")
  endif()
endforeach()
set(BE_FFT_SANITIZER none CACHE STRING "none, address, thread")
if(BE_FFT_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(BE_FFT_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
elseif(NOT BE_FFT_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown sanitizer")
endif()
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
configure_file("${BE_FFT_REFERENCE_SOURCE}/src/experimental/pv/fft.hpp" reference/fft.hpp COPYONLY)
add_library(fft_reference STATIC "${BE_FFT_REFERENCE_SOURCE}/src/experimental/pv/fft.cpp")
target_compile_definitions(fft_reference PRIVATE boiled_egg=fft_reference)
add_library(fft_candidate STATIC "${ROOT}/src/experimental/pv/fft.cpp")
find_package(Threads REQUIRED)
add_executable(fft_stage_controls controls.cpp)
target_include_directories(fft_stage_controls PRIVATE "${ROOT}/src/experimental/pv" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(fft_stage_controls PRIVATE fft_candidate fft_reference Threads::Threads)
enable_testing()
if(BE_FFT_SANITIZER STREQUAL "thread")
  add_test(NAME fft_stage_threads COMMAND fft_stage_controls threads)
else()
  foreach(name first_stage small_prefix large_prefix signed_zero immediate zero_budget noalloc threads)
    add_test(NAME fft_stage_${name} COMMAND fft_stage_controls ${name})
  endforeach()
endif()
