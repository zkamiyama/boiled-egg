cmake_minimum_required(VERSION 3.20)
project(boiled_egg_transport VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(BE_TRANSPORT_TESTS "Build native transport tests" ON)
option(BE_TRANSPORT_SANITIZE "ASan and UBSan" OFF)
option(BE_TRANSPORT_THREAD_SANITIZE "TSan independent-owner tests" OFF)
if(BE_TRANSPORT_SANITIZE AND BE_TRANSPORT_THREAD_SANITIZE)
  message(FATAL_ERROR "Address and thread sanitizers need separate builds")
endif()
if(BE_TRANSPORT_THREAD_SANITIZE)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
if(BE_TRANSPORT_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(PV_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../src/experimental/pv")
add_library(boiled_egg_transport SHARED transport.cpp "${PV_DIR}/fft.cpp")
target_include_directories(boiled_egg_transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${PV_DIR}")
target_compile_definitions(boiled_egg_transport PRIVATE BE_TRANSPORT_BUILD)
set_target_properties(boiled_egg_transport PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(NOT MSVC)
  target_compile_options(boiled_egg_transport PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
install(TARGETS boiled_egg_transport LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES transport.h transport.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/boiled_egg)
if(BE_TRANSPORT_TESTS)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(transport_threads_test test_threads.cpp)
  target_link_libraries(transport_threads_test PRIVATE boiled_egg_transport Threads::Threads)
  add_test(NAME transport_threads COMMAND transport_threads_test)
  add_executable(transport_test test_transport.cpp)
  target_link_libraries(transport_test PRIVATE boiled_egg_transport)
  if(UNIX)
    target_link_options(transport_test PRIVATE -Wl,--export-dynamic)
  endif()
  add_test(NAME transport_contract COMMAND transport_test)
  add_executable(transport_c_test test_c.c)
  target_link_libraries(transport_c_test PRIVATE boiled_egg_transport)
  add_test(NAME transport_c11 COMMAND transport_c_test)
endif()
