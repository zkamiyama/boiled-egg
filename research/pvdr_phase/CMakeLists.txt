cmake_minimum_required(VERSION 3.20)
project(boiled_egg_phase_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(PG_SANITIZER "none" CACHE STRING "none,address,thread")
if(PG_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(PG_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
elseif(NOT PG_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown sanitizer")
endif()
add_library(pvdr_phase SHARED phase.cpp)
target_compile_options(pvdr_phase PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
find_package(Threads REQUIRED)
add_executable(phase_test test.cpp)
target_link_libraries(phase_test PRIVATE pvdr_phase Threads::Threads)
enable_testing()
if(PG_SANITIZER STREQUAL "thread")
  add_test(NAME phase_threads COMMAND phase_test threads)
else()
  foreach(name magnitude zero gradients frequency_path temporal_path seed invalid budget repeat threads)
    add_test(NAME phase_${name} COMMAND phase_test ${name})
  endforeach()
endif()
if(NOT PG_SANITIZER STREQUAL "thread")
  add_executable(phase_c_test test_c.c)
  target_link_libraries(phase_c_test PRIVATE pvdr_phase)
  add_test(NAME phase_c_api COMMAND phase_c_test)
endif()
