cmake_minimum_required(VERSION 3.20)
project(boiled_egg_transient_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
add_library(transient_map SHARED bridge.cpp)
target_compile_features(transient_map PUBLIC cxx_std_20)
add_library(phase_heap SHARED ../phase_gradient/bridge.cpp)
target_compile_features(phase_heap PUBLIC cxx_std_20)
add_executable(transient_map_test test_map.cpp)
add_executable(phase_heap_test ../phase_gradient/test_kernel.cpp)
enable_testing()
add_test(NAME transient_map_test COMMAND transient_map_test)
add_test(NAME phase_heap_test COMMAND phase_heap_test)
