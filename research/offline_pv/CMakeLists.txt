cmake_minimum_required(VERSION 3.20)
project(boiled_egg_offline_pv_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OFFLINE_PV_SANITIZE "Use ASan and UBSan" OFF)
if(OFFLINE_PV_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(offline_pv_core STATIC offline.cpp "${ROOT}/src/experimental/pv/fft.cpp")
target_include_directories(offline_pv_core PUBLIC . PRIVATE "${ROOT}/src/experimental/pv")
target_compile_options(offline_pv_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(boiled_egg_offline_pv main.cpp "${ROOT}/tools/wav_io.cpp")
target_include_directories(boiled_egg_offline_pv PRIVATE "${ROOT}/tools")
target_link_libraries(boiled_egg_offline_pv PRIVATE offline_pv_core)
enable_testing()
add_executable(offline_pv_test test.cpp)
target_link_libraries(offline_pv_test PRIVATE offline_pv_core)
foreach(name identity length projection deterministic invalid)
  add_test(NAME offline_pv_${name} COMMAND offline_pv_test ${name})
endforeach()
