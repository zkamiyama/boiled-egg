cmake_minimum_required(VERSION 3.20)
project(boiled_egg_phase_edges LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ language standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PHASE_EDGES_SANITIZE "Instrument research kernels" OFF)
if(PHASE_EDGES_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(phase_edge SHARED bridge.cpp)
add_library(phase_heap SHARED ../phase_gradient/bridge.cpp)
add_executable(edge_kernel_test test_kernel.cpp)
add_executable(old_kernel_test ../phase_gradient/test_kernel.cpp)
foreach(target phase_edge phase_heap edge_kernel_test old_kernel_test)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
enable_testing()
add_test(NAME edge_curved_fields COMMAND edge_kernel_test)
add_test(NAME original_affine_fields COMMAND old_kernel_test)
add_executable(edge_kernel_bench bench_kernel.cpp)
