cmake_minimum_required(VERSION 3.25)
project(boiled_egg_formant_lab VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CLAP_ROOT "" CACHE PATH "Pinned CLAP 1.2.10 source directory")
set(VST3_ROOT "" CACHE PATH "Pinned VST3 SDK source directory")
option(FORMANT_LAB_VST3 "Build the experimental VST3 adapter" ON)
add_subdirectory(../cpp_pv_rt research_core)
if(NOT EXISTS "${CLAP_ROOT}/include/clap/clap.h")
  message(FATAL_ERROR "Set CLAP_ROOT to pinned CLAP source")
endif()
add_library(boiled_egg_formant_clap MODULE formant_clap.cpp)
target_include_directories(boiled_egg_formant_clap PRIVATE "${CLAP_ROOT}/include")
target_link_libraries(boiled_egg_formant_clap PRIVATE boiled_egg_research_host)
set_target_properties(boiled_egg_formant_clap PROPERTIES PREFIX "" SUFFIX ".clap")
if(FORMANT_LAB_VST3)
  if(NOT EXISTS "${VST3_ROOT}/pluginterfaces/vst/ivstaudioprocessor.h")
    message(FATAL_ERROR "Set VST3_ROOT to the pinned recursive SDK source")
  endif()
  set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
  set(SMTG_RUN_VST_VALIDATOR ON CACHE BOOL "" FORCE)
  set(SMTG_CREATE_MODULE_INFO ON CACHE BOOL "" FORCE)
  add_subdirectory("${VST3_ROOT}" vst3sdk)
  smtg_enable_vst3_sdk()
  smtg_add_vst3plugin(boiled_egg_formant_vst3 PACKAGE_NAME "boiled egg Formant Lab"
    SOURCES_LIST formant_vst3.cpp "${VST3_ROOT}/public.sdk/source/vst/vstsinglecomponenteffect.cpp")
  target_link_libraries(boiled_egg_formant_vst3 PRIVATE boiled_egg_research_host sdk)
endif()
if(BOILED_EGG_PV_RT_BUILD_TESTS)
  enable_testing()
  add_executable(formant_clap_test test_clap.cpp)
  target_include_directories(formant_clap_test PRIVATE "${CLAP_ROOT}/include")
  target_link_libraries(formant_clap_test PRIVATE boiled_egg_research_host ${CMAKE_DL_LIBS})
  target_link_options(formant_clap_test PRIVATE -Wl,--export-dynamic)
  add_test(NAME formant_clap_loaded COMMAND formant_clap_test $<TARGET_FILE:boiled_egg_formant_clap>)
  if(FORMANT_LAB_VST3)
    add_executable(formant_vst3_test test_vst3.cpp
      "${VST3_ROOT}/public.sdk/source/common/memorystream.cpp"
      "${VST3_ROOT}/public.sdk/source/vst/hosting/module.cpp"
      "${VST3_ROOT}/public.sdk/source/vst/hosting/module_linux.cpp")
    target_link_libraries(formant_vst3_test PRIVATE boiled_egg_research_host sdk_hosting sdk ${CMAKE_DL_LIBS})
    target_link_options(formant_vst3_test PRIVATE -Wl,--export-dynamic)
    add_test(NAME formant_vst3_loaded COMMAND formant_vst3_test "${CMAKE_CURRENT_BINARY_DIR}/VST3/Release/boiled egg Formant Lab.vst3")
  endif()
endif()
