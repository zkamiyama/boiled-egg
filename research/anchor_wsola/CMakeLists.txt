cmake_minimum_required(VERSION 3.20)
project(boiled_egg_anchor_research LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ANCHOR_SANITIZE "ASan/UBSan for the isolated offline kernel" OFF)
if(ANCHOR_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(anchor_wsola SHARED anchor.cpp)
target_compile_options(anchor_wsola PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
find_package(Threads REQUIRED)
add_executable(anchor_test test.cpp)
target_link_libraries(anchor_test PRIVATE anchor_wsola Threads::Threads)
add_executable(anchor_c_test test_c.c)
target_link_libraries(anchor_c_test PRIVATE anchor_wsola)
enable_testing()
foreach(name identity deterministic empty_marks ownership invalid threads)
  add_test(NAME anchor_${name} COMMAND anchor_test ${name})
endforeach()
add_test(NAME anchor_c COMMAND anchor_c_test)
