cmake_minimum_required(VERSION 3.20)
project(boiled_egg_nsdgt_foundation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(NSG_SANITIZER "none" CACHE STRING "none, address or thread")
if(NSG_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(NSG_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
elseif(NOT NSG_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown sanitizer")
endif()
add_library(nsdgt SHARED nsdgt.cpp ../../src/experimental/pv/fft.cpp)
target_include_directories(nsdgt PRIVATE ../../src/experimental/pv)
target_compile_options(nsdgt PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
find_package(Threads REQUIRED)
add_executable(nsg_test test.cpp)
target_link_libraries(nsg_test PRIVATE nsdgt Threads::Threads)
add_executable(nsg_c_test test_c.c)
target_link_libraries(nsg_c_test PRIVATE nsdgt)
enable_testing()
if(NSG_SANITIZER STREQUAL "thread")
  add_test(NAME nsg_threads COMMAND nsg_test threads)
else()
  foreach(name roundtrip direct_dft direct_synthesis basis edits linearity projection coverage invalid budget threads)
    add_test(NAME nsg_${name} COMMAND nsg_test ${name})
  endforeach()
  add_test(NAME nsg_c_api COMMAND nsg_c_test)
endif()
